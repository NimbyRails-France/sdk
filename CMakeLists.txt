cmake_minimum_required(VERSION 3.24)
project(NimbyRailsFranceSDK VERSION 0.8.0 LANGUAGES C CXX)

# Root orchestration only. Platform implementation and dependencies are selected
# once here, never injected into the shared engine or Kotlin-facing API.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(cmake/Architecture.cmake)
if(BUILD_TESTING)
    add_test(NAME architecture_boundaries COMMAND ${CMAKE_COMMAND} -DSDK_ROOT=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/Architecture.cmake)
endif()

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "NimbyRailsFranceSDK requires a 64-bit target")
endif()
if(WIN32)
    include(cmake/windows/Targets.cmake)
    include(cmake/windows/Tests.cmake)
    include(cmake/windows/Install.cmake)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(cmake/linux/Targets.cmake)
else()
    message(FATAL_ERROR "Supported SDK targets: Windows x64 and Linux x64 (observation research)")
endif()

# C/C++ assert-based tests must remain active in optimized configurations.
include(cmake/PortableTests.cmake)
if(BUILD_TESTING)
    get_property(sdk_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS sdk_targets)
        if(target MATCHES "(_tests|_test|_host)$")
            if(MSVC)
                target_compile_options(${target} PRIVATE /UNDEBUG)
            else()
                target_compile_options(${target} PRIVATE -UNDEBUG)
            endif()
        endif()
    endforeach()
endif()
