cmake_minimum_required(VERSION 3.24)
project(NrfPolicyOnlyHostVerification LANGUAGES CXX)
enable_testing()
if(NOT WIN32)
    message(FATAL_ERROR "This native adapter harness requires Windows.")
endif()
if(NOT POLICY_MOD OR NOT POLICY_SDK OR NOT EXISTS "${POLICY_MOD}" OR NOT EXISTS "${POLICY_SDK}")
    message(FATAL_ERROR "Set POLICY_MOD to the verified BC adapter DLL and POLICY_SDK to the SDK runtime DLL.")
endif()
get_filename_component(sdk_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(mod_directory "${POLICY_MOD}" DIRECTORY)
get_filename_component(mod_name "${POLICY_MOD}" NAME)
get_filename_component(mod_stem "${POLICY_MOD}" NAME_WE)
get_filename_component(runtime_directory "${POLICY_SDK}" DIRECTORY)
set(stage "${CMAKE_CURRENT_BINARY_DIR}/stage")
file(MAKE_DIRECTORY "${stage}")
file(COPY "${POLICY_MOD}" "${mod_directory}/${mod_stem}Kotlin.dll"
          "${mod_directory}/translations.json" DESTINATION "${stage}")
configure_file("${POLICY_SDK}" "${stage}/NrfPolicyHostActual.dll" COPYONLY)
file(GLOB runtime_support "${runtime_directory}/lib*.dll")
file(COPY ${runtime_support} DESTINATION "${stage}")

# Reuse the true packaged adapter. Only its SDK process/option boundary is
# synthetic; untouched imports forward to a renamed runtime in this sandbox.
execute_process(COMMAND "${CMAKE_OBJDUMP}" -p "${POLICY_MOD}" OUTPUT_VARIABLE imports COMMAND_ERROR_IS_FATAL ANY)
string(REGEX MATCHALL "NimbyInternal_[A-Za-z0-9_]+" sdk_imports "${imports}")
list(REMOVE_DUPLICATES sdk_imports)
set(overrides NimbyInternal_EnsureSignalUiBridge NimbyInternal_OpenProcess NimbyInternal_CaptureSessionSnapshot)
set(definition "EXPORTS\n")
foreach(symbol IN LISTS sdk_imports)
    if(NOT symbol IN_LIST overrides)
        string(APPEND definition "${symbol}=NrfPolicyHostActual.${symbol}\n")
    endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/facade.def" "${definition}")
add_library(policy_host_facade SHARED windows/fixture.cpp "${CMAKE_CURRENT_BINARY_DIR}/facade.def")
target_compile_features(policy_host_facade PRIVATE cxx_std_20)
target_include_directories(policy_host_facade PRIVATE "${sdk_root}/include")
target_compile_definitions(policy_host_facade PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(policy_host_facade PROPERTIES PREFIX "" OUTPUT_NAME NimbyRailsFranceSDK
    RUNTIME_OUTPUT_DIRECTORY "${stage}")
add_executable(policy_only_host_test windows/main.cpp)
target_compile_features(policy_only_host_test PRIVATE cxx_std_20)
target_compile_definitions(policy_only_host_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(policy_only_host_test PRIVATE psapi)
set_target_properties(policy_only_host_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${stage}")
add_dependencies(policy_only_host_test policy_host_facade)
if(MINGW)
    foreach(target policy_host_facade policy_only_host_test)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
    endforeach()
endif()
add_test(NAME policy_only_hosted_adapter COMMAND policy_only_host_test
    "$<TARGET_FILE:policy_host_facade>" "${stage}/${mod_name}")
file(MAKE_DIRECTORY "${stage}/profile")
set_tests_properties(policy_only_hosted_adapter PROPERTIES TIMEOUT 20 WORKING_DIRECTORY "${stage}"
    ENVIRONMENT "LOCALAPPDATA=${stage}/profile")
